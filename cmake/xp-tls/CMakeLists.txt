cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(SparkXpTls LANGUAGES C)
if(NOT MSVC OR MSVC_VERSION GREATER_EQUAL 1920 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the x86 v141_xp toolset.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_compile_definitions(_WIN32_WINNT=0x0501 WINVER=0x0501 WIN32_LEAN_AND_MEAN NOMINMAX
    MBEDTLS_USER_CONFIG_FILE="spark_xp_tls_config.h")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}")
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)
set(MBEDTLS_INCLUDE_DIR "${MBEDTLS_SOURCE}/include" CACHE PATH "" FORCE)
set(MBEDTLS_LIBRARY mbedtls CACHE STRING "" FORCE)
set(MBEDX509_LIBRARY mbedx509 CACHE STRING "" FORCE)
set(MBEDCRYPTO_LIBRARY mbedcrypto CACHE STRING "" FORCE)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(CURL_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(CURL_USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL OFF CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
set(ENABLE_THREADED_RESOLVER ON CACHE BOOL "" FORCE)
set(CURL_TARGET_WINDOWS_VERSION 0x0501 CACHE STRING "" FORCE)
add_subdirectory("${CURL_SOURCE}" curl)
install(FILES spark_xp_tls_config.h threading_alt.h DESTINATION include)
